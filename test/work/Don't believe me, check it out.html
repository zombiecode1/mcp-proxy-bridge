1. Server Health Check

Verify both Agent A (Proxi bridge, port 5001) and Agent B (MCP agent, port 3000) are running and responding.

# Check if both servers are running
curl -s http://localhost:5001/health | jq .
curl -s -o /dev/null -w "%{http_code}" -X HEAD http://localhost:3000/mcp

2. OAuth Token Flow

Test token issuance via JSON and form-urlencoded requests, plus error handling for invalid credentials and grant types.

# Get JWT token (JSON)
curl -s -X POST http://localhost:3000/auth/token \
  -H "Content-Type: application/json" \
  -d '{"grant_type":"client_credentials","client_id":"test-client","client_secret":"zombiecoder-dev-secret-2026"}'

# Get JWT token (form-urlencoded)
curl -s -X POST http://localhost:3000/auth/token \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -d 'grant_type=client_credentials&client_id=test-client&client_secret=zombiecoder-dev-secret-2026'

# Test invalid secret (should return 401)
curl -s -o /dev/null -w "%{http_code}" -X POST http://localhost:3000/auth/token \
  -H "Content-Type: application/json" \
  -d '{"grant_type":"client_credentials","client_id":"bad","client_secret":"wrong-secret"}'

# Test invalid grant type (should return 400)
curl -s -o /dev/null -w "%{http_code}" -X POST http://localhost:3000/auth/token \
  -H "Content-Type: application/json" \
  -d '{"grant_type":"password"}'

3. Agent B — Authenticated MCP Calls

Agent B (port 3000) requires OAuth Bearer tokens. Use the token from Section 2 to list tools, verify sessions, ping, and run the agent.

TOKEN="<paste your token here>"

# List tools
curl -s -X POST http://localhost:3000/mcp \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}'

# Verify session
curl -s -X POST http://localhost:3000/mcp \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":2,"method":"tools/call","params":{"name":"verify_session","arguments":{}}}'

# Ping agent
curl -s -X POST http://localhost:3000/mcp \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":3,"method":"tools/call","params":{"name":"ping_agent","arguments":{}}}'

# Run agent (with a prompt)
curl -s -X POST http://localhost:3000/mcp \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":4,"method":"tools/call","params":{"name":"run_agent","arguments":{"prompt":"Say hello and nothing else"}}}'

4. Agent A — Workspace Tools (No Auth)

Agent A (port 5001) uses session-based access (no OAuth). Test server info, MCP initialization, tool listing, workspace search, and indexing.

# Get server info
curl -s http://localhost:5001/mcp/info | jq .

# Initialize MCP session
curl -s -D - -X POST http://localhost:5001/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-03-26","clientInfo":{"name":"test","version":"1.0.0"},"capabilities":{}}}'

# List tools
curl -s -X POST http://localhost:5001/mcp \
  -H "Content-Type: application/json" \
  -H "Mcp-Session-Id: test-session" \
  -d '{"jsonrpc":"2.0","id":2,"method":"tools/list","params":{}}'

# Search workspace
curl -s -X POST http://localhost:5001/mcp \
  -H "Content-Type: application/json" \
  -H "Mcp-Session-Id: test-session" \
  -d '{"jsonrpc":"2.0","id":3,"method":"tools/call","params":{"name":"workspace_search","arguments":{"query":"test","maxResults":3}}}'

# Index workspace
curl -s -X POST http://localhost:5001/mcp \
  -H "Content-Type: application/json" \
  -H "Mcp-Session-Id: test-session" \
  -d '{"jsonrpc":"2.0","id":4,"method":"tools/call","params":{"name":"workspace_index","arguments":{"directory":"/tmp","maxFiles":100}}}'

9. Complete Verification Script

Single bash script that runs all health checks, OAuth, tool listing, SSE auth, CORS, DB integrity, and cross-isolation tests in sequence.

#!/usr/bin/env bash
# Comprehensive ZombieCoder verification
ERRORS=0

echo "=== 1. Server Health ==="
curl -sf http://localhost:5001/health > /dev/null && echo "  Proxi: OK" || { echo "  Proxi: FAIL"; ((ERRORS++)); }
curl -sf -X HEAD http://localhost:3000/mcp > /dev/null && echo "  Agent: OK" || { echo "  Agent: FAIL"; ((ERRORS++)); }

echo "=== 2. OAuth Token ==="
TOKEN=$(curl -sf -X POST http://localhost:3000/auth/token \
  -H "Content-Type: application/json" \
  -d '{"grant_type":"client_credentials","client_id":"verify","client_secret":"zombiecoder-dev-secret-2026"}' \
  | python3 -c "import sys,json; print(json.load(sys.stdin).get('access_token',''))" 2>/dev/null)
[ -n "$TOKEN" ] && echo "  Token: OK" || { echo "  Token: FAIL"; ((ERRORS++)); }

echo "=== 3. Agent B Tools ==="
TOOLS=$(curl -sf -X POST http://localhost:3000/mcp \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/list","params":{}}' \
  | python3 -c "import sys,json; d=json.load(sys.stdin); print(len(d.get('result',{}).get('tools',[])))")
[ "$TOOLS" -ge 3 ] && echo "  Agent B: $TOOLS tools OK" || { echo "  Agent B: FAIL"; ((ERRORS++)); }

echo "=== 4. Agent A Tools ==="
TOOLS_A=$(curl -sf http://localhost:5001/mcp/info | python3 -c "import sys,json; print(json.load(sys.stdin).get('tools',0))")
[ "$TOOLS_A" -ge 8 ] && echo "  Agent A: $TOOLS_A tools OK" || { echo "  Agent A: FAIL"; ((ERRORS++)); }

echo "=== 5. SSE Auth ==="
SSE=$(curl -sf -o /dev/null -w "%{http_code}" http://localhost:3000/sse/test 2>/dev/null || echo "401")
[ "$SSE" = "401" ] && echo "  SSE Auth: OK (returns $SSE)" || { echo "  SSE Auth: FAIL"; ((ERRORS++)); }

echo "=== 6. CORS ==="
CORS=$(curl -sf -D - http://localhost:3000/mcp -X HEAD 2>/dev/null | grep -ci "access-control-allow-origin" || echo "0")
[ "$CORS" -gt 0 ] && echo "  CORS: OK" || { echo "  CORS: FAIL"; ((ERRORS++)); }

echo "=== 7. Database ==="
DOCS=$(sqlite3 /home/sahon/mcp/proxi/.zombiecoder/state.db "SELECT COUNT(*) FROM rag_documents;" 2>/dev/null || echo "0")
CHUNKS=$(sqlite3 /home/sahon/mcp/proxi/.zombiecoder/state.db "SELECT COUNT(*) FROM rag_chunks;" 2>/dev/null || echo "0")
echo "  DB: $DOCS docs, $CHUNKS chunks"

echo ""
echo "Errors: $ERRORS"
[ "$ERRORS" -eq 0 ] && echo "ALL CHECKS PASSED" || echo "SOME CHECKS FAILED"
